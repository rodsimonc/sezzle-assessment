// Command server runs the calculator API and, when STATIC_DIR is set, the built frontend.
package main

import (
        "context"
        "flag"
        "fmt"
        "io"
        "log/slog"
        "net/http"
        "os"
        "os/signal"
        "strconv"
        "syscall"
        "time"

        "calculator/internal/config"
        "calculator/internal/server"
        "calculator/internal/version"
)

func main() {
        os.Exit(run(os.Args[1:], os.Getenv, os.Stdout, os.Stderr))
}

// run starts the service and returns the process exit code. SIGINT and SIGTERM end it cleanly.
func run(args []string, getenv func(string) string, stdout, stderr io.Writer) int {
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()
        return runContext(ctx, args, getenv, stdout, stderr)
}

func runContext(ctx context.Context, args []string, getenv func(string) string, stdout, stderr io.Writer) int {
        flags := flag.NewFlagSet("server", flag.ContinueOnError)
        flags.SetOutput(stderr)
        healthcheck := flags.Bool("healthcheck", false, "probe /healthz on 127.0.0.1 and exit 0 when it answers 200")
        if err := flags.Parse(args); err != nil {
                return 2
        }

        cfg, err := config.Load(getenv)
        if err != nil {
                _, _ = fmt.Fprintf(stderr, "config error: %v\n", err)
                return 1
        }

        if *healthcheck {
                if err := probe(ctx, "http://127.0.0.1:"+strconv.Itoa(cfg.Port)+"/healthz"); err != nil {
                        _, _ = fmt.Fprintf(stderr, "healthcheck failed: %v\n", err)
                        return 1
                }
                return 0
        }

        logger := newLogger(stdout, cfg)
        logger.Info("starting", "version", version.Version, "port", cfg.Port)
        if err := server.Run(ctx, cfg, logger); err != nil {
                logger.Error("server stopped", "error", err)
                return 1
        }
        return 0
}

func newLogger(w io.Writer, cfg config.Config) *slog.Logger {
        opts := &slog.HandlerOptions{Level: cfg.LogLevel}
        if cfg.LogFormat == "text" {
                return slog.New(slog.NewTextHandler(w, opts))
        }
        return slog.New(slog.NewJSONHandler(w, opts))
}

// probe sends GET to url with a 3 second limit and wants a 200.
func probe(ctx context.Context, url string) error {
        ctx, cancel := context.WithTimeout(ctx, 3*time.Second)
        defer cancel()
        req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, http.NoBody)
        if err != nil {
                return err
        }
        resp, err := http.DefaultClient.Do(req)
        if err != nil {
                return err
        }
        defer func() { _ = resp.Body.Close() }()
        if resp.StatusCode != http.StatusOK {
                return fmt.Errorf("status %d", resp.StatusCode)
        }
        return nil
}

package config

import (
        "fmt"
        "log/slog"
        "net/url"

// Config holds the validated settings.
type Config struct {
        Port               int
        StaticDir          string
        CORSAllowedOrigins []string
        RateLimitPerMinute int
        RateLimitBurst     int
        TrustProxy         bool
        LogLevel           slog.Level
        LogFormat          string
        ShutdownTimeout    time.Duration
}

// Load reads the settings through getenv. An unset or empty variable takes its default.
// A bad value returns an error that names the variable.
func Load(getenv func(string) string) (Config, error) {
        cfg := Config{
                StaticDir: getenv("STATIC_DIR"),
        }
        var err error
        if cfg.Port, err = intVar(getenv, "PORT", 8080, 1, 65535); err != nil {
                return Config{}, err
        }
        if cfg.RateLimitPerMinute, err = intVar(getenv, "RATE_LIMIT_PER_MINUTE", 120, 0, 1_000_000); err != nil {
                return Config{}, err
        }
        if cfg.RateLimitBurst, err = intVar(getenv, "RATE_LIMIT_BURST", 30, 1, 1_000_000); err != nil {
                return Config{}, err
        }
        seconds, err := intVar(getenv, "SHUTDOWN_TIMEOUT_SECONDS", 10, 1, 3600)
        if err != nil {
                return Config{}, err
        }
        cfg.ShutdownTimeout = time.Duration(seconds) * time.Second

        if cfg.TrustProxy, err = boolVar(getenv, "TRUST_PROXY", false); err != nil {
                return Config{}, err
        }
        if cfg.LogLevel, err = levelVar(getenv, "LOG_LEVEL"); err != nil {
                return Config{}, err
        }
        if cfg.LogFormat, err = formatVar(getenv, "LOG_FORMAT"); err != nil {
                return Config{}, err
        }
        if cfg.CORSAllowedOrigins, err = originsVar(getenv, "CORS_ALLOWED_ORIGINS"); err != nil {
                return Config{}, err
        }
        return cfg, nil
}

func intVar(getenv func(string) string, name string, def, lo, hi int) (int, error) {
        raw := strings.TrimSpace(getenv(name))
        if raw == "" {
                return def, nil
        }
        n, err := strconv.Atoi(raw)
        if err != nil || n < lo || n > hi {
                return 0, fmt.Errorf("%s must be a whole number from %d to %d, got %q", name, lo, hi, raw)
        }
        return n, nil
}

func boolVar(getenv func(string) string, name string, def bool) (bool, error) {
        raw := strings.TrimSpace(getenv(name))
        if raw == "" {
                return def, nil
        }
        b, err := strconv.ParseBool(raw)
        if err != nil {
                return false, fmt.Errorf("%s must be true or false, got %q", name, raw)
        }
        return b, nil
}

func levelVar(getenv func(string) string, name string) (slog.Level, error) {
        raw := strings.ToLower(strings.TrimSpace(getenv(name)))
        switch raw {
        case "", "info":
                return slog.LevelInfo, nil
        case "debug":
                return slog.LevelDebug, nil
        case "warn":
                return slog.LevelWarn, nil
        case "error":
                return slog.LevelError, nil
        }
        return 0, fmt.Errorf("%s must be debug, info, warn or error, got %q", name, raw)
}

func formatVar(getenv func(string) string, name string) (string, error) {
        raw := strings.ToLower(strings.TrimSpace(getenv(name)))
        switch raw {
        case "", "json":
                return "json", nil
        case "text":
                return "text", nil
        }
        return "", fmt.Errorf("%s must be json or text, got %q", name, raw)
}

// originsVar parses a comma-separated allowlist. Each entry is a bare origin such as https://app.example.com.
func originsVar(getenv func(string) string, name string) ([]string, error) {
        var origins []string
        for _, entry := range strings.Split(getenv(name), ",") {
                entry = strings.TrimSpace(entry)
                if entry == "" {
                        continue
                }
                u, err := url.Parse(entry)
                if err != nil || (u.Scheme != "http" && u.Scheme != "https") || u.Host == "" || u.Path != "" || u.RawQuery != "" || u.Fragment != "" || u.User != nil {
                        return nil, fmt.Errorf("%s entries must look like https://app.example.com, got %q", name, entry)
                }
                origins = append(origins, entry)
        }
        return origins, nil
}

// Codes for the failures the arithmetic itself can produce.
const (
        CodeDivisionByZero     = "DIVISION_BY_ZERO"
        CodeNegativeSquareRoot = "NEGATIVE_SQUARE_ROOT"
        CodeUndefinedResult    = "UNDEFINED_RESULT"
)

// Error is a business-rule failure. Message is a full sentence that is safe to show to a user.
type Error struct {
        Code    string
        Message string
}

func (e *Error) Error() string { return e.Message }

// The sentinels are shared and must be treated as read-only.
var (
        ErrDivisionByZero     = &Error{Code: CodeDivisionByZero, Message: "Division by zero is undefined."}
        ErrNegativeSquareRoot = &Error{Code: CodeNegativeSquareRoot, Message: "The square root of a negative number isn't a real number."}
        ErrUndefinedResult    = &Error{Code: CodeUndefinedResult, Message: "The result isn't a real number."}
)

// Package domain holds the pure arithmetic. It knows nothing about HTTP, rounding or request shapes.
package domain

import "math"

// Operation describes one calculator operation. Apply expects exactly Arity operands,
// and the service checks the count before it calls.
type Operation struct {
        Name        string
        Symbol      string
        Description string
        Arity       int
        Apply       func(operands []float64) (float64, error)
}

var operations = []Operation{
        {Name: "add", Symbol: "+", Arity: 2, Description: "Adds two numbers.", Apply: add},
        {Name: "subtract", Symbol: "-", Arity: 2, Description: "Subtracts the second number from the first.", Apply: subtract},
        {Name: "multiply", Symbol: "*", Arity: 2, Description: "Multiplies two numbers.", Apply: multiply},
        {Name: "divide", Symbol: "/", Arity: 2, Description: "Divides the first number by the second.", Apply: divide},
        {Name: "power", Symbol: "^", Arity: 2, Description: "Raises the first number to the power of the second.", Apply: power},
        {Name: "sqrt", Symbol: "sqrt", Arity: 1, Description: "Returns the square root of a non-negative number.", Apply: sqrt},
        {Name: "percentage", Symbol: "%", Arity: 2, Description: "Returns the first number percent of the second.", Apply: percentage},
        {Name: "reciprocal", Symbol: "1/x", Arity: 1, Description: "Returns 1 divided by the number.", Apply: reciprocal},
        {Name: "square", Symbol: "x^2", Arity: 1, Description: "Multiplies the number by itself.", Apply: square},
}

// All returns the operations in catalogue order. The slice is a copy.
func All() []Operation {
        out := make([]Operation, len(operations))
        copy(out, operations)
        return out
}

// Lookup finds an operation by its exact, case-sensitive name.
func Lookup(name string) (Operation, bool) {
        for _, op := range operations {
                if op.Name == name {
                        return op, true
                }
        }
        return Operation{}, false
}

func add(o []float64) (float64, error)      { return o[0] + o[1], nil }
func subtract(o []float64) (float64, error) { return o[0] - o[1], nil }
func multiply(o []float64) (float64, error) { return o[0] * o[1], nil }
func square(o []float64) (float64, error)   { return o[0] * o[0], nil }

func divide(o []float64) (float64, error) {
        if o[1] == 0 {
                return 0, ErrDivisionByZero
        }
        return o[0] / o[1], nil
}

func power(o []float64) (float64, error) {
        base, exp := o[0], o[1]
        if base == 0 && exp < 0 {
                return 0, ErrDivisionByZero
        }
        if base < 0 && exp != math.Trunc(exp) {
                return 0, ErrUndefinedResult
        }
        return math.Pow(base, exp), nil
}

func sqrt(o []float64) (float64, error) {
        if o[0] < 0 {
                return 0, ErrNegativeSquareRoot
        }
        return math.Sqrt(o[0]), nil
}

// percentage divides first so that 1e308 percent of 50 stays finite.
func percentage(o []float64) (float64, error) { return o[0] / 100 * o[1], nil }

func reciprocal(o []float64) (float64, error) {
        if o[0] == 0 {
                return 0, ErrDivisionByZero
        }
        return 1 / o[0], nil
}

import (

// badRequest is a decoding failure ready to be written as a Problem.
type badRequest struct {
        status int
        code   string
        detail string
        fields []problem.FieldError
}

func (b *badRequest) write(w http.ResponseWriter, r *http.Request) {
        problem.Write(w, r, b.status, b.code, b.detail, problem.WithErrors(b.fields...))
}

func invalidRequest(fields ...problem.FieldError) *badRequest {
        return &badRequest{status: http.StatusBadRequest, code: problem.CodeInvalidRequest, detail: problem.DetailInvalidRequest, fields: fields}
}

func invalidJSON() *badRequest {
        return &badRequest{status: http.StatusBadRequest, code: problem.CodeInvalidJSON, detail: problem.DetailInvalidJSON}
}

// readBody checks the media type (415), then reads the body within the size limit (413).
// The order is part of the contract: a 5000-byte text/plain body is a 415.
func readBody(r *http.Request) ([]byte, *badRequest) {
        mediaType, _, err := mime.ParseMediaType(r.Header.Get("Content-Type"))
        if err != nil || mediaType != "application/json" {
                return nil, &badRequest{status: http.StatusUnsupportedMediaType, code: problem.CodeUnsupportedMediaType, detail: problem.DetailUnsupportedMediaType}
        }
        body, err := io.ReadAll(r.Body)
        if err != nil {
                var tooLarge *http.MaxBytesError
                if errors.As(err, &tooLarge) {
                        return nil, &badRequest{status: http.StatusRequestEntityTooLarge, code: problem.CodePayloadTooLarge, detail: problem.DetailPayloadTooLarge}
                }
                return nil, invalidJSON()
        }
        return body, nil
}

// parseObject accepts exactly one JSON object and returns its members undecoded.
func parseObject(body []byte) (map[string]json.RawMessage, *badRequest) {
        dec := json.NewDecoder(bytes.NewReader(body))
        var value json.RawMessage
        if err := dec.Decode(&value); err != nil {
                return nil, invalidJSON()
        }
        var extra json.RawMessage
        if err := dec.Decode(&extra); !errors.Is(err, io.EOF) {
                return nil, invalidJSON()
        }
        var members map[string]json.RawMessage
        if value[0] != '{' || json.Unmarshal(value, &members) != nil {
                return nil, invalidRequest()
        }
        return members, nil
}

// decodeCalculation reads {operation, operands} and rejects anything else.
func decodeCalculation(body []byte) (CalculationRequest, *badRequest) {
        members, bad := parseObject(body)
        if bad != nil {
                return CalculationRequest{}, bad
        }
        var req CalculationRequest
        var fields []problem.FieldError
        var fe *problem.FieldError
        if req.Operation, fe = readOperation(members); fe != nil {
                fields = append(fields, *fe)
        }
        if req.Operands, fe = readOperands(members); fe != nil {
                fields = append(fields, *fe)
        }
        fields = append(fields, unknownFields(members, "operation", "operands")...)
        if len(fields) > 0 {
                return CalculationRequest{}, invalidRequest(fields...)
        }
        return req, nil
}

// decodeOperands reads {operands}. An operation key counts as unknown here, because the path names the operation.
func decodeOperands(body []byte) (OperandsRequest, *badRequest) {
        members, bad := parseObject(body)
        if bad != nil {
                return OperandsRequest{}, bad
        }
        var req OperandsRequest
        var fields []problem.FieldError
        var fe *problem.FieldError
        if req.Operands, fe = readOperands(members); fe != nil {
                fields = append(fields, *fe)
        }
        fields = append(fields, unknownFields(members, "operands")...)
        if len(fields) > 0 {
                return OperandsRequest{}, invalidRequest(fields...)
        }
        return req, nil
}

func readOperation(members map[string]json.RawMessage) (string, *problem.FieldError) {
        raw, present := members["operation"]
        if !present {
                return "", &problem.FieldError{Field: "operation", Message: "is required."}
        }
        var name string
        if isNull(raw) || json.Unmarshal(raw, &name) != nil {
                return "", &problem.FieldError{Field: "operation", Message: "must be a string."}
        }
        return name, nil
}

// readOperands needs an array of finite numbers. A string, null, boolean or nested value fails,
// and so does a number outside the float64 range.
func readOperands(members map[string]json.RawMessage) ([]float64, *problem.FieldError) {
        raw, present := members["operands"]
        if !present {
                return nil, &problem.FieldError{Field: "operands", Message: "is required."}
        }
        invalid := &problem.FieldError{Field: "operands", Message: "must be an array of numbers."}
        var items []json.RawMessage
        if isNull(raw) || json.Unmarshal(raw, &items) != nil {
                return nil, invalid
        }
        operands := make([]float64, len(items))
        for i, item := range items {
                if isNull(item) || json.Unmarshal(item, &operands[i]) != nil {
                        return nil, invalid
                }
        }
        return operands, nil
}

func isNull(raw json.RawMessage) bool { return bytes.Equal(raw, []byte("null")) }

// unknownFields lists every member outside known, sorted so the answer is deterministic.
func unknownFields(members map[string]json.RawMessage, known ...string) []problem.FieldError {
        var names []string
        for name := range members {
                if !contains(known, name) {
                        names = append(names, name)
                }
        }
        sort.Strings(names)
        fields := make([]problem.FieldError, len(names))
        for i, name := range names {
                fields[i] = problem.FieldError{Field: name, Message: "isn't allowed."}
        }
        return fields
}

func contains(list []string, s string) bool {
        for _, item := range list {
                if item == s {
                        return true
                }
        }
        return false
}

// Calculator is the part of the service the handlers use.
type Calculator interface {
        Calculate(name string, operands []float64) (service.Result, error)
        Operations() []domain.Operation
}

// Handler serves the API routes.
type Handler struct {
        calc   Calculator
        logger *slog.Logger
        spec   []byte
}

// New builds a Handler. spec is the OpenAPI document served at /api/v1/openapi.yaml.
func New(calc Calculator, logger *slog.Logger, spec []byte) *Handler {
        return &Handler{calc: calc, logger: logger, spec: spec}
}

// NotFound answers a request that matches no route.
func NotFound(w http.ResponseWriter, r *http.Request) {
        problem.Write(w, r, http.StatusNotFound, problem.CodeNotFound, problem.NotFoundDetail(r.Method, r.URL.Path))
}

// MethodNotAllowed answers a known path called with the wrong method. Go's own 405 is plain text,
// so every route checks the method itself.
func MethodNotAllowed(w http.ResponseWriter, r *http.Request, allowed ...string) {
        w.Header().Set("Allow", strings.Join(allowed, ", "))
        problem.Write(w, r, http.StatusMethodNotAllowed, problem.CodeMethodNotAllowed, problem.MethodNotAllowedDetail(r.Method, r.URL.Path, allowed))
}

func allow(w http.ResponseWriter, r *http.Request, method string) bool {
        if r.Method == method {
                return true
        }
        MethodNotAllowed(w, r, method)
        return false
}

// Calculations handles POST /api/v1/calculations.
func (h *Handler) Calculations(w http.ResponseWriter, r *http.Request) {
        if !allow(w, r, http.MethodPost) {
                return
        }
        body, bad := readBody(r)
        if bad != nil {
                bad.write(w, r)
                return
        }
        req, bad := decodeCalculation(body)
        if bad != nil {
                bad.write(w, r)
                return
        }
        h.calculate(w, r, req.Operation, req.Operands, false)
}

// RunOperation handles POST /api/v1/operations/{name}.
func (h *Handler) RunOperation(w http.ResponseWriter, r *http.Request) {
        if !allow(w, r, http.MethodPost) {
                return
        }
        body, bad := readBody(r)
        if bad != nil {
                bad.write(w, r)
                return
        }
        req, bad := decodeOperands(body)
        if bad != nil {
                bad.write(w, r)
                return
        }
        h.calculate(w, r, r.PathValue("name"), req.Operands, true)
}

// Operations handles GET /api/v1/operations.
func (h *Handler) Operations(w http.ResponseWriter, r *http.Request) {
        if !allow(w, r, http.MethodGet) {
                return
        }
        ops := h.calc.Operations()
        items := make([]OperationResponse, len(ops))
        for i, op := range ops {
                items[i] = OperationResponse{Name: op.Name, Symbol: op.Symbol, Arity: op.Arity, Description: op.Description}
        }
        h.writeJSON(w, r, OperationListResponse{Items: items, Total: len(items)})
}

// OpenAPI handles GET /api/v1/openapi.yaml.
func (h *Handler) OpenAPI(w http.ResponseWriter, r *http.Request) {
        if !allow(w, r, http.MethodGet) {
                return
        }
        w.Header().Set("Content-Type", "application/yaml")
        _, _ = w.Write(h.spec)
}

// Health handles GET /healthz.
func (h *Handler) Health(w http.ResponseWriter, r *http.Request) {
        if !allow(w, r, http.MethodGet) {
                return
        }
        h.writeJSON(w, r, healthResponse{Status: "ok", Version: version.Version})
}

func (h *Handler) calculate(w http.ResponseWriter, r *http.Request, name string, operands []float64, fromPath bool) {
        res, err := h.calc.Calculate(name, operands)
        if err != nil {
                h.fail(w, r, err, fromPath)
                return
        }
        h.writeJSON(w, r, CalculationResponse{Operation: res.Operation, Operands: res.Operands, Result: res.Value})
}

// fail maps a service error to a Problem. An unknown name is a 422 in a body and a 404 in the path.
func (h *Handler) fail(w http.ResponseWriter, r *http.Request, err error, fromPath bool) {
        var unknown *service.UnknownOperationError
        var count *service.OperandCountError
        var rule *domain.Error
        switch {
        case errors.As(err, &unknown) && fromPath:
                problem.Write(w, r, http.StatusNotFound, problem.CodeUnknownOperation, problem.UnknownOperationDetail(unknown.Name))
        case errors.As(err, &unknown):
                problem.Write(w, r, http.StatusUnprocessableEntity, problem.CodeUnknownOperation, problem.UnknownOperationDetail(unknown.Name),
                        problem.WithErrors(problem.FieldError{Field: "operation", Message: h.validNames()}))
        case errors.As(err, &count):
                problem.Write(w, r, http.StatusBadRequest, problem.CodeInvalidOperandCount, problem.DetailInvalidOperandCount,
                        problem.WithErrors(problem.FieldError{Field: "operands", Message: count.Error() + "."}))
        case errors.As(err, &rule):
                problem.Write(w, r, http.StatusUnprocessableEntity, rule.Code, rule.Message)
        default:
                h.internalError(w, r, err)
        }
}

func (h *Handler) validNames() string {
        ops := h.calc.Operations()
        names := make([]string, len(ops))
        for i, op := range ops {
                names[i] = op.Name
        }
        return "must be one of: " + strings.Join(names, ", ") + "."
}

func (h *Handler) internalError(w http.ResponseWriter, r *http.Request, err error) {
        h.logger.Error("request failed", "error", err, "requestId", problem.RequestID(r.Context()))
        problem.Write(w, r, http.StatusInternalServerError, problem.CodeInternalError, problem.DetailInternalError)
}

func (h *Handler) writeJSON(w http.ResponseWriter, r *http.Request, v any) {
        body, err := json.Marshal(v)
        if err != nil {
                h.internalError(w, r, err)
                return
        }
        w.Header().Set("Content-Type", "application/json")
        _, _ = w.Write(body)
}

// Package middleware holds the HTTP middleware. Each one has the shape func(http.Handler) http.Handler
// (or a function that returns one), so the server composes them in a fixed order.
package middleware

import (

        "encoding/hex"

// MaxBodyBytes is the request body limit for /api/ routes.
const MaxBodyBytes = 4096

const requestIDHeader = "X-Request-Id"

const contentSecurityPolicy = "default-src 'self'; script-src 'self'; style-src 'self'; img-src 'self' data:; " +
        "connect-src 'self'; font-src 'self'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'"

var validRequestID = regexp.MustCompile(`^[A-Za-z0-9._-]{1,64}$`)

// IsAPIPath reports whether a path belongs to the API (/api or anything under /api/).
func IsAPIPath(path string) bool {
        return path == "/api" || strings.HasPrefix(path, "/api/")
}

// RequestID gives every request an id. A valid incoming X-Request-Id is kept. Anything else is replaced
// with 16 random bytes as 32 hex characters.
func RequestID(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                id := r.Header.Get(requestIDHeader)
                if !validRequestID.MatchString(id) {
                        id = newRequestID()
                }
                w.Header().Set(requestIDHeader, id)
                next.ServeHTTP(w, r.WithContext(problem.WithRequestID(r.Context(), id)))
        })
}

func newRequestID() string {
        b := make([]byte, 16)
        _, _ = rand.Read(b) // never fails since Go 1.24
        return hex.EncodeToString(b)
}

// Recover turns a panic into a 500 Problem. The body never carries the panic value or the stack.
// Those go to the log.
func Recover(logger *slog.Logger) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        defer func() {
                                v := recover()
                                if v == nil {
                                        return
                                }
                                if err, isErr := v.(error); isErr && errors.Is(err, http.ErrAbortHandler) {
                                        panic(v) // net/http uses this value to abort a response quietly
                                }
                                logger.Error("panic recovered", "panic", v, "stack", string(debug.Stack()), "requestId", problem.RequestID(r.Context()))
                                problem.Write(w, r, http.StatusInternalServerError, problem.CodeInternalError, problem.DetailInternalError)
                        }()
                        next.ServeHTTP(w, r)
                })
        }
}

// SecurityHeaders sets the fixed hardening headers on every response and no-store on API and health responses.
func SecurityHeaders(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                h := w.Header()
                h.Set("X-Content-Type-Options", "nosniff")
                h.Set("X-Frame-Options", "DENY")
                h.Set("Referrer-Policy", "no-referrer")
                h.Set("Content-Security-Policy", contentSecurityPolicy)
                if IsAPIPath(r.URL.Path) || r.URL.Path == "/healthz" {
                        h.Set("Cache-Control", "no-store")
                }
                next.ServeHTTP(w, r)
        })
}

// BodyLimit caps API request bodies. It wraps the body instead of answering itself, so the handler
// can check the media type first (415) and report an oversize body (413) when it reads.
func BodyLimit(limit int64) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        if !IsAPIPath(r.URL.Path) {
                                next.ServeHTTP(w, r)
                                return
                        }
                        limited := http.MaxBytesReader(w, r.Body, limit)
                        r.Body = limited
                        next.ServeHTTP(w, r)
                        // net/http drains an unread body itself, but only when it owns the reader. A handler that
                        // answers 415 or 405 early would leave the bytes unread, and closing the socket on unread
                        // bytes makes Windows reset the connection before the client reads the response.
                        _, _ = io.Copy(io.Discard, limited)
                })
        }
}

// CORS answers preflights and tags responses for origins on the allowlist. With an empty list it does nothing,
// so no response carries Access-Control-Allow-Origin.
func CORS(allowed []string) func(http.Handler) http.Handler {
        set := make(map[string]struct{}, len(allowed))
        for _, origin := range allowed {
                set[origin] = struct{}{}
        }
        return func(next http.Handler) http.Handler {
                if len(set) == 0 {
                        return next
                }
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        h := w.Header()
                        h.Add("Vary", "Origin")
                        origin := r.Header.Get("Origin")
                        if origin == "" {
                                next.ServeHTTP(w, r)
                                return
                        }
                        _, ok := set[origin]
                        preflight := r.Method == http.MethodOptions && r.Header.Get("Access-Control-Request-Method") != ""
                        switch {
                        case preflight && !ok:
                                problem.Write(w, r, http.StatusForbidden, problem.CodeCORSOriginNotAllowed, problem.DetailCORSOriginNotAllowed)
                        case preflight:
                                h.Set("Access-Control-Allow-Origin", origin)
                                h.Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
                                h.Set("Access-Control-Allow-Headers", "Content-Type, X-Request-Id")
                                h.Set("Access-Control-Max-Age", "600")
                                w.WriteHeader(http.StatusNoContent)
                        default:
                                if ok {
                                        h.Set("Access-Control-Allow-Origin", origin)
                                        h.Set("Access-Control-Expose-Headers", "X-Request-Id, Retry-After")
                                }
                                next.ServeHTTP(w, r)
                        }
                })
        }
}

// statusWriter records the status and the byte count for the access log.
type statusWriter struct {
        http.ResponseWriter
        status int
        bytes  int
}

func (w *statusWriter) WriteHeader(status int) {
        if w.status == 0 {
                w.status = status
        }
        w.ResponseWriter.WriteHeader(status)
}

func (w *statusWriter) Write(b []byte) (int, error) {
        if w.status == 0 {
                w.status = http.StatusOK
        }
        n, err := w.ResponseWriter.Write(b)
        w.bytes += n
        return n, err
}

// Unwrap lets http.ResponseController reach the real writer for flushing and deadlines.
func (w *statusWriter) Unwrap() http.ResponseWriter { return w.ResponseWriter }

// Logging writes one slog record per request. A panic that passes through still gets its record, as a 500.
func Logging(logger *slog.Logger) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()
                        sw := &statusWriter{ResponseWriter: w}
                        finished := false
                        defer func() {
                                status := sw.status
                                switch {
                                case !finished:
                                        status = http.StatusInternalServerError
                                case status == 0:
                                        status = http.StatusOK
                                }
                                logger.LogAttrs(r.Context(), slog.LevelInfo, "request",
                                        slog.String("method", r.Method),
                                        slog.String("path", r.URL.Path),
                                        slog.Int("status", status),
                                        slog.Int64("durationMs", time.Since(start).Milliseconds()),
                                        slog.Int("bytes", sw.bytes),
                                        slog.String("requestId", problem.RequestID(r.Context())),
                                )
                        }()
                        next.ServeHTTP(sw, r)
                        finished = true
                })
        }
}

// RateLimitOptions configures RateLimit. A PerMinute of 0 turns the limiter off.
type RateLimitOptions struct {
        PerMinute  int
        Burst      int
        TrustProxy bool
        // Now is the clock. It defaults to time.Now and exists so tests can drive time.
        Now func() time.Time
        // SweepEvery is how often the janitor evicts idle buckets. It defaults to one minute.
        SweepEvery time.Duration
}

type bucket struct {
        tokens float64
        last   time.Time
}

// limiter is a token bucket per client key. Tokens refill at rate per second up to burst.
type limiter struct {
        mu      sync.Mutex
        buckets map[string]*bucket
        rate    float64
        burst   float64
        now     func() time.Time
}

func newLimiter(opts RateLimitOptions) *limiter {
        now := opts.Now
        if now == nil {
                now = time.Now
        }
        return &limiter{
                buckets: make(map[string]*bucket),
                rate:    float64(opts.PerMinute) / 60,
                burst:   float64(opts.Burst),
                now:     now,
        }
}

// allow takes one token for key. When none is left it returns the wait in whole seconds, at least 1.
func (l *limiter) allow(key string) (bool, int) {
        l.mu.Lock()
        defer l.mu.Unlock()
        now := l.now()
        b, ok := l.buckets[key]
        if !ok {
                b = &bucket{tokens: l.burst, last: now}
                l.buckets[key] = b
        }
        if elapsed := now.Sub(b.last).Seconds(); elapsed > 0 {
                b.tokens = math.Min(l.burst, b.tokens+elapsed*l.rate)
        }
        b.last = now
        if b.tokens >= 1 {
                b.tokens--
                return true, 0
        }
        // Fewer than 1 token means the wait is positive, so rounding up gives at least 1 second.
        return false, int(math.Ceil((1 - b.tokens) / l.rate))
}

// sweep drops buckets that have been idle long enough to refill completely. A refilled bucket
// equals a fresh one, so dropping it changes no answer.
func (l *limiter) sweep() {
        l.mu.Lock()
        defer l.mu.Unlock()
        now := l.now()
        refill := time.Duration(l.burst / l.rate * float64(time.Second))
        for key, b := range l.buckets {
                if now.Sub(b.last) >= refill {
                        delete(l.buckets, key)
                }
        }
}

func (l *limiter) size() int {
        l.mu.Lock()
        defer l.mu.Unlock()
        return len(l.buckets)
}

// janitor sweeps on a ticker until ctx ends.
func (l *limiter) janitor(ctx context.Context, every time.Duration) {
        ticker := time.NewTicker(every)
        defer ticker.Stop()
        for {
                select {
                case <-ctx.Done():
                        return
                case <-ticker.C:
                        l.sweep()
                }
        }
}

// RateLimit guards /api/ routes with a token bucket per client. Health checks and static files pass freely.
// The janitor goroutine stops when ctx ends.
func RateLimit(ctx context.Context, opts RateLimitOptions) func(http.Handler) http.Handler {
        if opts.PerMinute <= 0 {
                return func(next http.Handler) http.Handler { return next }
        }
        l := newLimiter(opts)
        every := opts.SweepEvery
        if every <= 0 {
                every = time.Minute
        }
        go l.janitor(ctx, every)
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        if !IsAPIPath(r.URL.Path) {
                                next.ServeHTTP(w, r)
                                return
                        }
                        ok, wait := l.allow(clientKey(r, opts.TrustProxy))
                        if ok {
                                next.ServeHTTP(w, r)
                                return
                        }
                        w.Header().Set("Retry-After", strconv.Itoa(wait))
                        problem.Write(w, r, http.StatusTooManyRequests, problem.CodeRateLimited, problem.RateLimitedDetail(wait), problem.WithRetryAfter(wait))
                })
        }
}

// clientKey is the remote address. With a trusted proxy it's the first X-Forwarded-For entry,
// as long as that entry is an IP address.
func clientKey(r *http.Request, trustProxy bool) string {
        if trustProxy {
                first, _, _ := strings.Cut(r.Header.Get("X-Forwarded-For"), ",")
                if addr, err := netip.ParseAddr(strings.TrimSpace(first)); err == nil {
                        return addr.String()
                }
        }
        host, _, err := net.SplitHostPort(r.RemoteAddr)
        if err != nil {
                return r.RemoteAddr
        }
        return host
}

// FieldError names one invalid field.
type FieldError struct {
        Field   string `json:"field"`
        Message string `json:"message"`
}

// Problem is the response body for every error.
type Problem struct {
        Type       string       `json:"type"`
        Title      string       `json:"title"`
        Status     int          `json:"status"`
        Detail     string       `json:"detail"`
        Instance   string       `json:"instance"`
        Code       string       `json:"code"`
        RequestID  string       `json:"requestId"`
        Errors     []FieldError `json:"errors,omitempty"`
        RetryAfter int          `json:"retryAfter,omitempty"`
}

// Option adjusts a Problem before it's written.
type Option func(*Problem)

// WithErrors attaches field-level errors.
func WithErrors(errs ...FieldError) Option {
        return func(p *Problem) { p.Errors = errs }
}

// WithRetryAfter sets the whole seconds a rate-limited client should wait.
func WithRetryAfter(seconds int) Option {
        return func(p *Problem) { p.RetryAfter = seconds }
}

// titles is fixed so a Go upgrade can't change the wording of http.StatusText.
var titles = map[int]string{
        http.StatusBadRequest:            "Bad Request",
        http.StatusForbidden:             "Forbidden",
        http.StatusNotFound:              "Not Found",
        http.StatusMethodNotAllowed:      "Method Not Allowed",
        http.StatusRequestEntityTooLarge: "Content Too Large",
        http.StatusUnsupportedMediaType:  "Unsupported Media Type",
        http.StatusUnprocessableEntity:   "Unprocessable Entity",
        http.StatusTooManyRequests:       "Too Many Requests",
        http.StatusInternalServerError:   "Internal Server Error",
}

// Title returns the fixed title for a status. A status outside the table falls back to the standard text.
func Title(status int) string {
        if t, ok := titles[status]; ok {
                return t
        }
        return http.StatusText(status)
}

type requestIDKey struct{}

        id, _ := ctx.Value(requestIDKey{}).(string)
        return id
}

// Write sends a Problem. Instance comes from the request path and requestId from the request context.
func Write(w http.ResponseWriter, r *http.Request, status int, code, detail string, opts ...Option) {
        p := Problem{
                Type:      "about:blank",
                Title:     Title(status),
                Status:    status,
                Detail:    detail,
                Instance:  r.URL.Path,
                Code:      code,
                RequestID: RequestID(r.Context()),
        }
        for _, opt := range opts {
                opt(&p)
        }
        w.Header().Set("Content-Type", "application/problem+json")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(p)
}

package server

// New builds the handler with its full middleware chain. The rate limiter's janitor goroutine
// lives as long as the process. Run ties it to its context instead.
func New(cfg config.Config, logger *slog.Logger) http.Handler {
        return build(context.Background(), cfg, logger, time.Now)
}

func build(ctx context.Context, cfg config.Config, logger *slog.Logger, now func() time.Time) http.Handler {
        return withMiddleware(ctx, routes(cfg, logger), cfg, logger, now)
}

func routes(cfg config.Config, logger *slog.Logger) http.Handler {
        h := handler.New(service.New(), logger, api.Spec)

        mux := http.NewServeMux()
        mux.HandleFunc("/api/v1/calculations", h.Calculations)
        mux.HandleFunc("/api/v1/operations", h.Operations)
        mux.HandleFunc("/api/v1/operations/{name}", h.RunOperation)
        mux.HandleFunc("/api/v1/openapi.yaml", h.OpenAPI)
        mux.HandleFunc("/healthz", h.Health)
        mux.HandleFunc("/api", handler.NotFound)
        mux.HandleFunc("/api/", handler.NotFound)
        static := newStatic(cfg.StaticDir, logger)
        mux.Handle("/", static)
        return withoutRedirects(mux, static)
}

// withoutRedirects answers paths with dot segments or doubled slashes itself. ServeMux would send
// those a 307 to the cleaned path. The static handler cleans the path on its own and never leaves
// the site directory, and an API path like that is a plain 404.
func withoutRedirects(mux, static http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                p := r.URL.Path
                if p == "" || r.Method == http.MethodConnect || cleanPath(p) == p {
                        mux.ServeHTTP(w, r)
                        return
                }
                if p == "/api" || strings.HasPrefix(p, "/api/") {
                        handler.NotFound(w, r)
                        return
                }
                static.ServeHTTP(w, r)
        })
}

// cleanPath mirrors how ServeMux cleans a path: path.Clean plus the trailing slash.
func cleanPath(p string) string {
        cleaned := path.Clean(p)
        if p[len(p)-1] == '/' && cleaned != "/" {
                cleaned += "/"
        }
        return cleaned
}

// withMiddleware wraps next in the fixed chain. Outermost first: RequestID, Recover, SecurityHeaders,
// Logging, CORS, RateLimit, BodyLimit. Every response, including the ones the inner layers write
// themselves, carries the request id and the security headers.
func withMiddleware(ctx context.Context, next http.Handler, cfg config.Config, logger *slog.Logger, now func() time.Time) http.Handler {
        rateLimit := middleware.RateLimit(ctx, middleware.RateLimitOptions{
                PerMinute:  cfg.RateLimitPerMinute,
                Burst:      cfg.RateLimitBurst,
                TrustProxy: cfg.TrustProxy,
                Now:        now,
        })
        chained := middleware.BodyLimit(middleware.MaxBodyBytes)(next)
        chained = rateLimit(chained)
        chained = middleware.CORS(cfg.CORSAllowedOrigins)(chained)
        chained = middleware.Logging(logger)(chained)
        chained = middleware.SecurityHeaders(chained)
        chained = middleware.Recover(logger)(chained)
        return middleware.RequestID(chained)
}

// Run listens on cfg.Port and serves until ctx ends, then shuts down within cfg.ShutdownTimeout.
func Run(ctx context.Context, cfg config.Config, logger *slog.Logger) error {
        var lc net.ListenConfig
        ln, err := lc.Listen(ctx, "tcp", ":"+strconv.Itoa(cfg.Port))
        if err != nil {
                return fmt.Errorf("listen: %w", err)
        }
        return Serve(ctx, ln, cfg, logger)
}

// Serve runs the server on an existing listener. It returns nil after a clean shutdown.
func Serve(ctx context.Context, ln net.Listener, cfg config.Config, logger *slog.Logger) error {
        srv := &http.Server{
                Handler:           build(ctx, cfg, logger, time.Now),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
                ErrorLog:          slog.NewLogLogger(logger.Handler(), slog.LevelError),
        }
        failed := make(chan error, 1)
        go func() { failed <- srv.Serve(ln) }()

        logger.Info("server listening", "addr", ln.Addr().String(), "staticDir", cfg.StaticDir)
        select {
        case err := <-failed:
                return fmt.Errorf("serve: %w", err)
        case <-ctx.Done():
        }

        logger.Info("shutting down", "timeoutSeconds", int(cfg.ShutdownTimeout.Seconds()))
        shutdownCtx, cancel := context.WithTimeout(context.Background(), cfg.ShutdownTimeout)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return fmt.Errorf("shutdown: %w", err)
        }
        if err := <-failed; !errors.Is(err, http.ErrServerClosed) {
                return fmt.Errorf("serve: %w", err)
        }
        return nil
}

// staticDirExists reports whether dir is a directory, so New can warn about a bad STATIC_DIR.
func staticDirExists(dir string) bool {
        info, err := os.Stat(dir)
        return err == nil && info.IsDir()
}

const (
        indexFile       = "index.html"
        assetsPrefix    = "/assets/"
        cacheImmutable  = "public, max-age=31536000, immutable"
        cacheRevalidate = "no-cache"
)

// contentTypes covers what a Vite build emits. Windows reads MIME types from the registry,
// which can map .js to text/plain, so the common ones are fixed here.
var contentTypes = map[string]string{
        ".html":        "text/html; charset=utf-8",
        ".js":          "text/javascript; charset=utf-8",
        ".mjs":         "text/javascript; charset=utf-8",
        ".css":         "text/css; charset=utf-8",
        ".json":        "application/json",
        ".map":         "application/json",
        ".webmanifest": "application/manifest+json",
        ".txt":         "text/plain; charset=utf-8",
        ".svg":         "image/svg+xml",
        ".png":         "image/png",
        ".jpg":         "image/jpeg",
        ".ico":         "image/x-icon",
        ".woff2":       "font/woff2",
}

// newStatic serves the built frontend from dir. An empty dir makes the service API-only, so every
// non-API path is a 404 Problem.
func newStatic(dir string, logger *slog.Logger) http.Handler {
        if dir == "" {
                return http.HandlerFunc(handler.NotFound)
        }
        if !staticDirExists(dir) {
                logger.Warn("STATIC_DIR is not a directory, static files will answer 404", "staticDir", dir)
        }
        return &staticHandler{files: os.DirFS(dir)}
}

type staticHandler struct {
        files fs.FS
}

func (s *staticHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
        if r.Method != http.MethodGet && r.Method != http.MethodHead {
                handler.MethodNotAllowed(w, r, http.MethodGet, http.MethodHead)
                return
        }
        // Cleaning an absolute path removes every ".." before the name reaches the file system.
        // os.DirFS also refuses names with backslashes or drive colons on Windows.
        clean := path.Clean("/" + r.URL.Path)
        name := strings.TrimPrefix(clean, "/")
        if name == "" {
                name = indexFile
        }
        if s.serve(w, r, name, clean) {
                return
        }
        // A missing hashed asset is a real 404. Serving the page for it would hide a broken deploy.
        if strings.HasPrefix(clean, assetsPrefix) || !s.serve(w, r, indexFile, "/"+indexFile) {
                handler.NotFound(w, r)
        }
}

// serve writes the file called name and reports whether it found a regular file.
func (s *staticHandler) serve(w http.ResponseWriter, r *http.Request, name, urlPath string) bool {
        f, err := s.files.Open(name)
        if err != nil {
                return false
        }
        defer func() { _ = f.Close() }()
        info, err := f.Stat()
        if err != nil || info.IsDir() {
                return false
        }
        content, ok := f.(io.ReadSeeker)
        if !ok {
                return false
        }
        h := w.Header()
        if ct, known := contentTypes[strings.ToLower(path.Ext(name))]; known {
                h.Set("Content-Type", ct)
        } else if ct := mime.TypeByExtension(path.Ext(name)); ct != "" {
                h.Set("Content-Type", ct)
        }
        if strings.HasPrefix(urlPath, assetsPrefix) {
                h.Set("Cache-Control", cacheImmutable)
        } else {
                h.Set("Cache-Control", cacheRevalidate)
        }
        http.ServeContent(w, r, name, info.ModTime(), content)
        return true
}

// OperandCountError reports an operand list of the wrong length.

}

func (e *OperandCountError) Error() string {
        noun := "operands"
        if e.Want == 1 {
                noun = "operand"
        }
        return fmt.Sprintf("%s needs %d %s, got %d", e.Operation, e.Want, noun, e.Got)
}

// Result is one finished calculation.
type Result struct {
        Operation string
        Operands  []float64
        Value     float64
}

// Calculator runs operations. It keeps no state between calls. lookup is a field so a test can
// add an operation that returns NaN, which no real operation can.
type Calculator struct {
        lookup func(name string) (domain.Operation, bool)
}

// New returns a Calculator.
func New() *Calculator { return &Calculator{lookup: domain.Lookup} }

// Operations lists the catalogue in its fixed order.
func (*Calculator) Operations() []domain.Operation { return domain.All() }

// Calculate validates the request, runs the operation and cleans up the result.
func (c *Calculator) Calculate(name string, operands []float64) (Result, error) {
        op, ok := c.lookup(name)
        if !ok {
                return Result{}, &UnknownOperationError{Name: name}
        }
        if len(operands) != op.Arity {
                return Result{}, &OperandCountError{Operation: name, Want: op.Arity, Got: len(operands)}
        }
        value, err := op.Apply(operands)
        if err != nil {
                return Result{}, err
        }
        switch {
        case math.IsNaN(value):
                return Result{}, domain.ErrUndefinedResult
        case math.IsInf(value, 0):
                return Result{}, errResultOutOfRange
        }
        return Result{Operation: name, Operands: operands, Value: clean(value)}, nil
}

// clean rounds to 15 significant digits and turns negative zero into zero.
// Rounding can push a value just under MaxFloat64 over the edge. ParseFloat then
// reports a range error and the unrounded value stays.
func clean(x float64) float64 {
        rounded, err := strconv.ParseFloat(strconv.FormatFloat(x, 'g', 15, 64), 64)
        if err != nil {
                rounded = x
        }
        if rounded == 0 {
                return 0
        }
        return rounded
}
